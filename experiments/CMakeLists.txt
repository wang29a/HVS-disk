function(experiment group name source)
    string(MAKE_C_IDENTIFIER "experiment_${group}_${name}" target)
    add_executable("${target}" "${source}" ${ARGN})
    set_target_properties("${target}" PROPERTIES
        OUTPUT_NAME "${name}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/experiments/${group}")
    set(LAST_EXPERIMENT "${target}" PARENT_SCOPE)
endfunction()

function(update_library target folder)
    add_library("${target}" STATIC
        "${folder}/update_store.cpp" "${folder}/deg_prune.cpp" "${folder}/dual_update.cpp")
    target_include_directories("${target}" PUBLIC "${folder}")
    target_link_libraries("${target}" PUBLIC pipeann Threads::Threads)
endfunction()

# Give each checked-in executable a namespaced target. Helper translation
# units and generated variants have explicit owners.
file(GLOB_RECURSE archived_cpp CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "*.cpp")
foreach(file IN LISTS archived_cpp)
    if(file MATCHES "(^|/)(runtime|generated|build)/" OR
       file MATCHES "/(odin_driver|driver|author_driver|replay_author|native_search)\\.cpp$")
        continue()
    endif()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${file}" contents)
    if(NOT contents MATCHES "int[ \t\r\n]+main[ \t\r\n]*\\(")
        continue()
    endif()
    string(REGEX MATCH "^[^/]+" group "${file}")
    get_filename_component(name "${file}" NAME_WE)
    experiment("${group}" "${name}" "${file}")
    if(group STREQUAL V1 OR group STREQUAL IO2 OR group STREQUAL V3 OR group STREQUAL V4)
        foreach(version IN ITEMS V1 IO2 V3 V4)
            if(group STREQUAL ${version})
                target_link_libraries("${LAST_EXPERIMENT}" PRIVATE "update_${version}")
            endif()
        endforeach()
    elseif(group STREQUAL "update-e1-beam-v3")
        target_link_libraries("${LAST_EXPERIMENT}" PRIVATE update_V3)
    elseif(group STREQUAL "update-perf-audit-2026-11-17")
        target_link_libraries("${LAST_EXPERIMENT}" PRIVATE update_V1)
    elseif(group STREQUAL METRIC)
        target_link_libraries("${LAST_EXPERIMENT}" PRIVATE metric_hybrid metric_aux pipeann)
    elseif(file MATCHES "/replay_b\\.cpp$")
        target_link_libraries("${LAST_EXPERIMENT}" PRIVATE backlink_b metric_aux pipeann)
    elseif(file MATCHES "/test_actual_pruner\\.cpp$")
        target_link_libraries("${LAST_EXPERIMENT}" PRIVATE unified metric_aux pipeann)
    else()
        target_link_libraries("${LAST_EXPERIMENT}" PRIVATE pipeann Threads::Threads)
    endif()
    if(group STREQUAL V3 AND name STREQUAL "static_query")
        target_compile_definitions("${LAST_EXPERIMENT}" PRIVATE
            HVS_ORIGINAL_QUERY_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/multigraph-20261113/scripts/multigraph.cpp")
    endif()
    if(name MATCHES "^test_|_check$|^check_")
        target_compile_options("${LAST_EXPERIMENT}" PRIVATE -UNDEBUG)
    endif()
    if(file MATCHES "/merge_check\\.cpp$")
        target_compile_definitions("${LAST_EXPERIMENT}" PRIVATE HVS_MERGE_V4)
    endif()
endforeach()
